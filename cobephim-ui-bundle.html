Vue 3 / Nuxt 3 Component Cheatsheet

Mẫu code component sẵn sàng copy trực tiếp vào dự án Vue/Nuxt của bạn.

components/MovieCard.vue (Thẻ phim với badge 4K & Sub/Dub) 
<template>
  <div class="sw-item group relative transition-all duration-300 hover:-translate-y-1">
    <NuxtLink :to="'/phim/' + movie.slug" class="v-thumbnail block relative w-full pb-[145%] rounded-xl overflow-hidden bg-[#2f3346] shadow-lg">
      <!-- Badges -->
      <span v-if="movie.is4k" class="thumb-4k-badge absolute top-1.5 right-1.5 z-10 px-1.5 py-0.5 text-[10px] font-extrabold rounded text-black bg-gradient-to-tr from-[#ffd875] to-white shadow">4K</span>
      <span v-else-if="movie.isBilingual" class="thumb-bilingual-badge absolute top-1.5 right-1.5 z-10 px-1.5 py-0.5 text-[10px] font-bold rounded text-[#07140c] bg-gradient-to-tr from-[#1a8f4a] to-[#60fc90] shadow">Song Ngữ</span>

      <div class="pin-new absolute bottom-0 inset-x-0 z-10 flex gap-1 p-1.5 bg-gradient-to-t from-black/80 to-transparent">
        <span v-if="movie.hasSub" class="line-pd text-[9px] font-bold px-1.5 py-0.5 rounded bg-white/20 text-white">P.Đề</span>
        <span v-if="movie.hasDub" class="line-tm text-[9px] font-bold px-1.5 py-0.5 rounded bg-[#2ca35d] text-white">T.Minh</span>
        <span v-if="movie.hasVoice" class="line-lt text-[9px] font-bold px-1.5 py-0.5 rounded bg-[#1667cf] text-white">L.Tiếng</span>
      </div>

      <img :src="movie.poster" :alt="movie.name" class="absolute inset-0 w-full h-full object-cover transition-transform duration-500 group-hover:scale-105" loading="lazy" />
    </NuxtLink>

    <div class="info pt-2.5">
      <h4 class="item-title text-sm font-semibold text-white truncate group-hover:text-[#ffd875] transition-colors">
        <NuxtLink :to="'/phim/' + movie.slug">{{ movie.name }}</NuxtLink>
      </h4>
      <div class="alias-title text-xs text-white/50 truncate mt-0.5">{{ movie.originName }}</div>
    </div>
  </div>
</template>

<script setup lang="ts">
defineProps<{
  movie: {
    name: string
    originName: string
    slug: string
    poster: string
    is4k?: boolean
    isBilingual?: boolean
    hasSub?: boolean
    hasDub?: boolean
    hasVoice?: boolean
  }
}>()
</script>

defineProps<{

components/ServerPickerLed.vue (Server Picker với hiệu ứng viền LED chạy) 
<template>
  <div class="episode-server-dropdown__led-ring relative inline-block min-w-[210px] rounded-xl">
    <!-- SVG Running Border LED -->
    <svg class="episode-server-dropdown__led-svg absolute inset-0 w-full h-full pointer-events-none z-0 overflow-visible" viewBox="0 0 210 44" preserveAspectRatio="none">
      <rect class="episode-server-dropdown__led-track fill-none stroke-white/10 stroke-[1.5]" x="1" y="1" width="208" height="42" rx="10" ry="10" />
      <rect class="episode-server-dropdown__led-chase fill-none stroke-[#ffe9a8] stroke-[2.25] [stroke-linecap:round] [stroke-dasharray:8_92] [animation:episode-server-led-chase_2.2s_linear_infinite] drop-shadow-[0_0_5px_rgba(255,210,90,0.7)]" x="1" y="1" width="208" height="42" rx="10" ry="10" pathLength="100" />
    </svg>

    <button @click="isOpen = !isOpen" class="episode-server-dropdown__toggle relative z-10 flex items-center justify-between w-full h-11 px-3.5 rounded-[10px] text-sm font-semibold text-white/95 bg-white/10 hover:bg-white/15 border border-white/15 transition-colors">
      <span>⚡ {{ currentServer }}</span>
      <span class="text-xs transition-transform" :class="{ 'rotate-180': isOpen }">▼</span>
    </button>

    <!-- Dropdown Menu -->
    <div v-if="isOpen" class="absolute top-[calc(100%+6px)] left-0 right-0 z-50 bg-[#191b24] border border-white/15 rounded-xl shadow-2xl overflow-hidden py-1">
      <div v-for="srv in servers" :key="srv" @click="currentServer = srv; isOpen = false" class="px-3.5 py-2 text-sm text-white/80 hover:bg-white/10 hover:text-[#ffd875] cursor-pointer">
        ⚡ {{ srv }}
      </div>
    </div>
  </div>
</template>

<script setup lang="ts">
import { ref } from 'vue'
const isOpen = ref(false)
const currentServer = ref('Vietsub #1 (VIP Fast)')
const servers = ['Vietsub #1 (VIP Fast)', 'Vietsub #2 (Dự phòng)', 'Thuyết Minh #1', 'Lồng Tiếng HD']
</script>

<style scoped>
@keyframes episode-server-led-chase {
  100% { stroke-dashoffset: -100; }
}
</style>
